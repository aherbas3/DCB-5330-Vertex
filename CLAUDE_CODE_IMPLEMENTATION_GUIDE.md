-- Create appointments table
CREATE TABLE appointments (
id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
provider_id BIGINT NOT NULL REFERENCES providers(id) ON DELETE CASCADE,
user_id TEXT NOT NULL,  -- Firebase UID
user_email VARCHAR NOT NULL,  -- For easier queries

-- Appointment details
appointment_date DATE NOT NULL,
start_time TIME NOT NULL,
end_time TIME NOT NULL,
appointment_type VARCHAR(50) NOT NULL,  -- e.g., "consultation", "follow-up", "check-up"

-- Status tracking
status VARCHAR(50) NOT NULL DEFAULT 'scheduled',  -- scheduled, confirmed, completed, cancelled, no-show
cancellation_reason TEXT,
cancelled_at TIMESTAMP,

-- Metadata
notes TEXT,
created_at TIMESTAMP DEFAULT NOW(),
updated_at TIMESTAMP DEFAULT NOW(),

-- Constraints
CONSTRAINT valid_time CHECK (start_time < end_time),
CONSTRAINT valid_status CHECK (status IN ('scheduled', 'confirmed', 'completed', 'cancelled', 'no-show'))
);

-- Indexes for performance
CREATE INDEX idx_appointments_provider_id ON appointments(provider_id);
CREATE INDEX idx_appointments_user_id ON appointments(user_id);
CREATE INDEX idx_appointments_appointment_date ON appointments(appointment_date);
CREATE INDEX idx_appointments_status ON appointments(status);
CREATE INDEX idx_appointments_user_date ON appointments(user_id, appointment_date);

-- Enable RLS for real-time notifications
ALTER TABLE appointments ENABLE ROW LEVEL SECURITY;

-- RLS Policy: Users can view their own appointments
CREATE POLICY "Users can view own appointments"
ON appointments FOR SELECT
USING (auth.uid()::TEXT = user_id OR auth.jwt() ->> 'email' = user_email);

-- RLS Policy: Users can only modify their own appointments
CREATE POLICY "Users can modify own appointments"
ON appointments FOR UPDATE
USING (auth.uid()::TEXT = user_id);

-- RLS Policy: Users can insert their own appointments
CREATE POLICY "Users can insert own appointments"
ON appointments FOR INSERT
WITH CHECK (auth.uid()::TEXT = user_id);

CREATE TABLE provider_slots (
id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
provider_id BIGINT NOT NULL REFERENCES providers(id) ON DELETE CASCADE,

-- Recurring slot configuration
day_of_week SMALLINT NOT NULL,  -- 0=Sunday, 6=Saturday
start_time TIME NOT NULL,
end_time TIME NOT NULL,
slot_duration SMALLINT NOT NULL DEFAULT 30,  -- in minutes

-- Slot availability
is_active BOOLEAN DEFAULT true,
date_from DATE,  -- Optional: specific date range
date_to DATE,

created_at TIMESTAMP DEFAULT NOW(),
updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_provider_slots_provider_id ON provider_slots(provider_id);
CREATE INDEX idx_provider_slots_day_of_week ON provider_slots(day_of_week);



CREATE TABLE notifications (
id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
user_id TEXT NOT NULL,
user_email VARCHAR NOT NULL,
appointment_id BIGINT REFERENCES appointments(id) ON DELETE CASCADE,

notification_type VARCHAR(50) NOT NULL,  -- 'appointment_confirmed', 'appointment_reminder', 'appointment_cancelled'
title VARCHAR NOT NULL,
message TEXT NOT NULL,

-- Delivery status
is_sent BOOLEAN DEFAULT false,
sent_at TIMESTAMP,
delivery_method VARCHAR(50),  -- 'sms', 'email', 'push'

created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_notifications_user_id ON notifications(user_id);
CREATE INDEX idx_notifications_appointment_id ON notifications(appointment_id);
